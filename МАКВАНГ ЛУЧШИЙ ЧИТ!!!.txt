local function isFriend(pl)
for _,fid in ipairs(g.Friends) do
if fid==pl.UserId then return true end
end
return false
end

local function toggleFriend(pl)
local found=false
for i,fid in ipairs(g.Friends) do
if fid==pl.UserId then
table.remove(g.Friends,i)
found=true
break
end
end
if not found then
g.Friends[#g.Friends+1]=pl.UserId
end
saveG()
end

local function getNearestPlayer()
local cc2=l.Character
local chrp=cc2 and cc2:FindFirstChild("HumanoidRootPart")
if not chrp then return nil end
local nearest=nil
local nearestDist=g.AutoRange
for _,pl in ipairs(p:GetPlayers()) do
if pl~=l and pl.Parent and not isFriend(pl) then
local pchr=pl.Character and pl.Character:FindFirstChild("HumanoidRootPart")
if pchr then
local d=(pchr.Position-chrp.Position).Magnitude
if d<nearestDist then
nearestDist=d
nearest=pl
end
end
end
end
return nearest
end

local function ragLoop()
pcall(function()
local cc=l.Character
if not cc or not cc.Parent then return end
local hum=cc:FindFirstChildOfClass("Humanoid")
if hum then
local st=hum:GetState()
if st==Enum.HumanoidStateType.Physics or st==Enum.HumanoidStateType.Ragdoll or st==Enum.HumanoidStateType.FallingDown then
hum:ChangeState(Enum.HumanoidStateType.Running)
if workspace.CurrentCamera then workspace.CurrentCamera.CameraSubject=hum end
pcall(function()
local pm=l.PlayerScripts:FindFirstChild("PlayerModule")
if pm then require(pm:FindFirstChild("ControlModule")):Enable() end
end)
end
end
for _,obj in ipairs(cc:GetDescendants()) do
if obj:IsA("Motor6D") and not obj.Enabled then obj.Enabled=true end
end
end)
end

local function stopMagnet()
g.MagnetOn=false
if magnetConn then magnetConn:Disconnect() magnetConn=nil end
if antiRagConn then antiRagConn:Disconnect() antiRagConn=nil end
removeHighlight()
pcall(function()
local cc=l.Character
local hum=cc and cc:FindFirstChildOfClass("Humanoid")
if hum then hum.WalkSpeed=16 end
end)
g.Target=nil
end

local function startMagnet(tp)
if not tp or not tp.Parent then return end
stopMagnet()
g.Target=tp
g.MagnetOn=true
addHighlight(tp)
antiRagConn=r.Heartbeat:Connect(ragLoop)
magnetConn=r.Heartbeat:Connect(function(dt)
if not g.MagnetOn then return end
pcall(function()
local tc=tp.Character
if not tc then stopMagnet() return end
if highlight and highlight.Parent~=tc then addHighlight(tp) end
local thrp=tc:FindFirstChild("HumanoidRootPart")
local cc2=l.Character
local chrp=cc2 and cc2:FindFirstChild("HumanoidRootPart")
if not thrp or not chrp then return end
local dir=thrp.Position-chrp.Position
local dist=dir.Magnitude
if dist<1.5 then return end
local step=MAGNET_SPEED*dt
if step>dist then step=dist end
local newPos=chrp.Position+dir.Unit*step
chrp.CFrame=CFrame.new(newPos,newPos+dir.Unit)
end)
end)
end

local function stopAutoMagnet()
g.AutoMagnet=false
if autoMagnetConn then autoMagnetConn:Disconnect() autoMagnetConn=nil end
if autoRagConn then autoRagConn:Disconnect() autoRagConn=nil end
removeCircle()
if autoBtn then autoBtn.BackgroundColor3=T.BTN end
pcall(function()
local cc=l.Character
local hum=cc and cc:FindFirstChildOfClass("Humanoid")
if hum then hum.WalkSpeed=16 end
end)
end

local function startAutoMagnet()
g.AutoMagnet=true
if not antiRagConn and not manualRagConn then
autoRagConn=r.Heartbeat:Connect(ragLoop)
end
if g.ShowCircle then
createCircle()
end
if autoBtn then autoBtn.BackgroundColor3=T.ACC end
autoMagnetConn=r.Heartbeat:Connect(function(dt)
if not g.AutoMagnet then return end
pcall(function()
local cc=l.Character
local chrp=cc and cc:FindFirstChild("HumanoidRootPart")
if not chrp then return end
local nearest=getNearestPlayer()
if not nearest then
if radiusCircle then removeCircle() end
return
end
local nc=nearest.Character
local nhrp=nc and nc:FindFirstChild("HumanoidRootPart")
if not nhrp then
if radiusCircle then removeCircle() end
return
end
local dir=nhrp.Position-chrp.Position
local dist=dir.Magnitude
if dist<1.5 then return end
if radiusCircle and not radiusCircle.Parent then createCircle() end
local step=MAGNET_SPEED*dt
if step>dist then step=dist end
local newPos=chrp.Position+dir.Unit*step
chrp.CFrame=CFrame.new(newPos,newPos+dir.Unit)
if radiusCircle and radiusCircle.Parent then
radiusCircle.CFrame=CFrame.new(chrp.Position-Vector3.new(0,3,0))*CFrame.Angles(0,0,math.rad(90))
end
end)
end)
end

local function playersListChanged()
local now={}
for _,pl in ipairs(p:GetPlayers()) do
if pl~=l and pl.Parent then now[pl]=true end
end
local changed=false
for pl,_ in pairs(now) do if not lastPlayerList[pl] then changed=true break end end
for pl,_ in pairs(lastPlayerList) do if not now[pl] then changed=true break end end
lastPlayerList=now
return changed
end

fpos=f.Position
end
end)

local d=i.Position-fstart
f.Position=UDim2.new(fpos.X.Scale,fpos.X.Offset+d.X,fpos.Y.Scale,fpos.Y.Offset+d.Y)
end
end)

end)

if d.Magnitude>5 then moved=true end
b.Position=UDim2.new(bpos.X.Scale,bpos.X.Offset+d.X,bpos.Y.Scale,bpos.Y.Offset+d.Y)
end
end)

u.InputEnded:Connect(function(i)
if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then bdrag=false end
end)

local open=false
local animating=false
local function toggle()
if animating then return end
animating=true
open=not open
pcall(function()
if open then
f.Visible=true
f.BackgroundTransparency=1
local tw=tsv:Create(f,TweenInfo.new(0.2),{BackgroundTransparency=0})
tw:Play()
tw.Completed:Wait()
else
local tw=tsv:Create(f,TweenInfo.new(0.2),{BackgroundTransparency=1})
tw:Play()
tw.Completed:Wait()
f.Visible=false
end
end)
animating=false
end

if not moved then toggle() end
end)
