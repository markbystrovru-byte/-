tb.Text=""
tb.ZIndex=8

tb.Active=true
tb.Parent=tab.page
Instance.new("UICorner",tb).CornerRadius=UDim.new(1,0)
local fl=Instance.new("Frame")
fl.Size=UDim2.new((def-min)/(max-min),0,1,0)
fl.BackgroundColor3=C_ACC
fl.BorderSizePixel=0
fl.ZIndex=9
fl.Parent=tb
Instance.new("UICorner",fl).CornerRadius=UDim.new(1,0)
tab.setY(Y+26)
local dr=false
local function upd(x)
local rel=math.clamp((x-tb.AbsolutePosition.X+0.5)/math.max(tb.AbsoluteSize.X,1),0,1)
fl.Size=UDim2.new(rel,0,1,0)
local v=min+rel*(max-min)
lb.Text=label..": "..string.format("%.2f",v)
if cb then pcall(cb,v) end
end
tb.InputBegan:Connect(function(i)
if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then dr=true; upd(i.Position.X) end
end)
u.InputChanged:Connect(function(i)
if dr and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then upd(i.Position.X) end
end)
u.InputEnded:Connect(function(i)
if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then dr=false end
end)
end

local origSizes={}
local function saveSizes()
origSizes={}
if c then
for _,v in ipairs(c:GetDescendants()) do
if v:IsA("BasePart") then origSizes[v]=v.Size end
end
end
end
saveSizes()
local randScaleOn=false
local function applyScale(k)
pcall(function()
if not c then return end
if next(origSizes)==nil then saveSizes() end
for part,sz in pairs(origSizes) do
if part.Parent then part.Size=sz*k end
end
local hum=c:FindFirstChildOfClass("Humanoid")
if hum then
pcall(function()
local desc=hum:GetAppliedDescription()
desc.HeightScale=math.clamp(k,0.1,5)
desc.WidthScale=math.clamp(k,0.1,5)
desc.DepthScale=math.clamp(k,0.1,5)
hum:ApplyDescription(desc)
end)
end
end)
end
